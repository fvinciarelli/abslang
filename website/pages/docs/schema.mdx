Every ABS version has a normative JSON Schema. Implementations use it to validate documents before parsing.

## Current schema (v0.1)

```json
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "$id": "https://abs-lang.org/schema/v0.1/abs.schema.json",
  "title": "ABS Document",
  "type": "object",
  "required": ["session", "behaviors"],
  "properties": {
    "session": {
      "type": "string",
      "description": "Human-readable session name"
    },
    "description": {
      "type": "string"
    },
    "abs_version": {
      "type": "string",
      "enum": ["0.1"]
    },
    "behaviors": {
      "type": "array",
      "items": { "$ref": "#/$defs/behavior" }
    },
    "evaluations": {
      "type": "array",
      "items": { "$ref": "#/$defs/evaluation" }
    }
  },
  "$defs": {
    "behavior": {
      "type": "object",
      "required": ["actor", "action"],
      "properties": {
        "actor": { "type": "string" },
        "action": { "type": "string" },
        "target": { "type": "string" },
        "content": {},
        "capture": {
          "type": "object",
          "additionalProperties": { "type": "string" }
        },
        "with": { "type": "object" },
        "evaluations": {
          "type": "array",
          "items": { "$ref": "#/$defs/evaluation" }
        }
      }
    },
    "evaluation": {
      "type": "object",
      "minProperties": 1,
      "maxProperties": 1
    }
  }
}
```

## Using the schema

### In your ABS document

```yaml
# yaml-language-server: $schema=https://abs-lang.org/schema/v0.1/abs.schema.json
session: My session
behaviors:
  - actor: user
    action: says
    content: "Hello"
```

### Programmatic validation

```javascript
import Ajv from 'ajv';
const schema = await fetch('https://abs-lang.org/schema/v0.1/abs.schema.json').then(r => r.json());
const ajv = new Ajv();
const validate = ajv.compile(schema);
if (!validate(absDocument)) {
  console.error(validate.errors);
}
```
